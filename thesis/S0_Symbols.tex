% LIST OF SYMBOLS
%--------------------------------------------------------------------%

\begin{listofsymbols}
\begin{description}

\item{\makebox[1.1in][l]{$\mathbb{R}, \mathbb{R}^d$}} \quad Real line, $d$-dimensional Euclidean space
\item{\makebox[1.1in][l]{$\mathbb{R}_+$}} \quad Set of non-negative real numbers $[0, \infty)$
\item{\makebox[1.1in][l]{$\mathbb{Q}$}} \quad Set of rational numbers
\item{\makebox[1.1in][l]{$\mathbb{N}$}} \quad Set of natural numbers $\{1, 2, 3, \dots\}$
\item{\makebox[1.1in][l]{$\mathbf{1}, I_d$}} \quad Vector of all ones, $d \times d$ identity matrix
\item{\makebox[1.1in][l]{$\lVert \cdot \rVert$}} \quad Standard Euclidean norm ($\ell_2$-norm)
\item{\makebox[1.1in][l]{$\langle \cdot, \cdot \rangle$}} \quad Standard Euclidean inner product
\item{\makebox[1.1in][l]{$\mathbbm{1}_{\{\cdot\}}, \mathbbm{1}_A$}} \quad Indicator function of an event or set $A$

\vspace{0.6em}

\item{\makebox[1.1in][l]{$\mathbb{P}(\cdot)$}} \quad Probability measure of an event
\item{\makebox[1.1in][l]{$\mathbb{E}[\cdot]$}} \quad Mathematical expectation operator
\item{\makebox[1.1in][l]{$\mathrm{Var}(\cdot)$}} \quad Variance of a random variable
\item{\makebox[1.1in][l]{$\mathrm{Cov}(\cdot, \cdot)$}} \quad Covariance between random variables

\vspace{0.6em}

\item{\makebox[1.1in][l]{$\mathcal{O}(\cdot), \widetilde{\mathcal{O}}(\cdot)$}} \quad Big-O asymptotic notation, Big-O hiding logarithmic factors
\item{\makebox[1.1in][l]{$\mathcal{H}_k, \|\cdot\|_{\mathcal{H}_k}$}} \quad Reproducing Kernel Hilbert Space (RKHS) and RKHS norm for kernel $k$

\end{description}
\end{listofsymbols}
